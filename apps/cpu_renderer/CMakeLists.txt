add_executable(cpu_renderer main.cpp)
target_link_libraries(cpu_renderer PRIVATE rt::common)

set(_rt_oidn_default ON)
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(_rt_oidn_default OFF)
endif()
option(RT_ENABLE_OIDN "Enable Intel Open Image Denoise for the CPU renderer" ${_rt_oidn_default})
if(RT_ENABLE_OIDN)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "OIDN's official Windows package is 64-bit, but this build uses a ${CMAKE_SIZEOF_VOID_P}-byte pointer toolchain. Reconfigure with a 64-bit compiler, or use -DRT_ENABLE_OIDN=OFF.")
  endif()
  # Prefer the repository-local official binary package, while allowing a normal
  # system/package-manager installation to be selected when this folder is absent.
  if(NOT OpenImageDenoise_DIR AND EXISTS "${PROJECT_SOURCE_DIR}/third_party/oidn/oidn-2.5.1.x64.windows/lib/cmake/OpenImageDenoise-2.5.1")
    set(OpenImageDenoise_DIR "${PROJECT_SOURCE_DIR}/third_party/oidn/oidn-2.5.1.x64.windows/lib/cmake/OpenImageDenoise-2.5.1")
  endif()
  find_package(OpenImageDenoise CONFIG REQUIRED)
  if(TARGET OpenImageDenoise)
    target_link_libraries(cpu_renderer PRIVATE OpenImageDenoise)
  elseif(TARGET OpenImageDenoise::OpenImageDenoise)
    target_link_libraries(cpu_renderer PRIVATE OpenImageDenoise::OpenImageDenoise)
  else()
    message(FATAL_ERROR "OpenImageDenoise was found, but its library target is missing")
  endif()
  target_compile_definitions(cpu_renderer PRIVATE RT_HAS_OIDN=1)
  if(WIN32 AND EXISTS "${PROJECT_SOURCE_DIR}/third_party/oidn/oidn-2.5.1.x64.windows/bin")
    set(_oidn_bin "${PROJECT_SOURCE_DIR}/third_party/oidn/oidn-2.5.1.x64.windows/bin")
    foreach(_dll OpenImageDenoise.dll OpenImageDenoise_core.dll OpenImageDenoise_device_cpu.dll tbb12.dll tbbbind_2_5.dll)
      if(EXISTS "${_oidn_bin}/${_dll}")
        add_custom_command(TARGET cpu_renderer POST_BUILD
          COMMAND ${CMAKE_COMMAND} -E copy_if_different "${_oidn_bin}/${_dll}" "$<TARGET_FILE_DIR:cpu_renderer>/${_dll}")
      endif()
    endforeach()
  endif()
endif()

find_package(OpenMP)
if(OpenMP_CXX_FOUND)
  target_link_libraries(cpu_renderer PRIVATE OpenMP::OpenMP_CXX)
endif()
